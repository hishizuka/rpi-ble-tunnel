cmake_minimum_required(VERSION 3.16)
project(rpi-ble-tunnel VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)

add_library(rpi_ble_tunnel_echo pi/echo.c pi/bridge.c pi/mux.c)
target_include_directories(rpi_ble_tunnel_echo PUBLIC pi)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0 gio-unix-2.0)
    find_library(BLUETOOTH_LIBRARY bluetooth)
    if(NOT BLUETOOTH_LIBRARY)
        message(FATAL_ERROR "libbluetooth is required (Debian: libbluetooth-dev)")
    endif()
    add_executable(rpi-ble-tunneld pi/rpi-ble-tunneld.c)
    target_link_libraries(rpi-ble-tunneld PRIVATE rpi_ble_tunnel_echo PkgConfig::GIO ${BLUETOOTH_LIBRARY})
    install(TARGETS rpi-ble-tunneld RUNTIME DESTINATION bin)
    install(FILES LICENSE DESTINATION share/doc/rpi-ble-tunnel)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(echo_test tests/echo_test.c)
    target_compile_options(echo_test PRIVATE -UNDEBUG)
    target_link_libraries(echo_test PRIVATE rpi_ble_tunnel_echo)
    add_test(NAME packet_echo COMMAND echo_test)
    add_executable(bridge_test tests/bridge_test.c)
    target_compile_options(bridge_test PRIVATE -UNDEBUG)
    target_link_libraries(bridge_test PRIVATE rpi_ble_tunnel_echo)
    add_test(NAME duplex_bridge COMMAND bridge_test)
    add_executable(mux_test tests/mux_test.c)
    target_compile_options(mux_test PRIVATE -UNDEBUG)
    target_link_libraries(mux_test PRIVATE rpi_ble_tunnel_echo)
    add_test(NAME multiplex_protocol COMMAND mux_test)
    add_executable(socks_queue_test tests/socks_queue_test.c)
    target_compile_options(socks_queue_test PRIVATE -UNDEBUG)
    add_test(NAME socks_queue_bounds COMMAND socks_queue_test)
    add_executable(mux_wait_test tests/mux_wait_test.c)
    target_compile_options(mux_wait_test PRIVATE -UNDEBUG)
    add_test(NAME multiplex_event_wait COMMAND mux_wait_test)
    add_executable(mux_test_server tests/mux_test_server.c)
    target_link_libraries(mux_test_server PRIVATE rpi_ble_tunnel_echo)
    set_tests_properties(packet_echo duplex_bridge multiplex_protocol socks_queue_bounds multiplex_event_wait PROPERTIES TIMEOUT 20)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(gatt_identity_test tests/gatt_identity_test.c)
        target_compile_options(gatt_identity_test PRIVATE -UNDEBUG)
        target_link_libraries(gatt_identity_test PRIVATE rpi_ble_tunnel_echo PkgConfig::GIO ${BLUETOOTH_LIBRARY})
        add_test(NAME gatt_hostname_identity COMMAND gatt_identity_test)
        set_tests_properties(gatt_hostname_identity PROPERTIES TIMEOUT 20)
        add_executable(mux_glib_test tests/mux_glib_test.c)
        target_compile_options(mux_glib_test PRIVATE -UNDEBUG)
        target_link_libraries(mux_glib_test PRIVATE rpi_ble_tunnel_echo PkgConfig::GIO ${BLUETOOTH_LIBRARY})
        add_test(NAME multiplex_glib_event_wait COMMAND mux_glib_test)
        set_tests_properties(multiplex_glib_event_wait PROPERTIES TIMEOUT 20)
    endif()
endif()
